struct Pt {
  dbl x, y, z;
  Pt() : x(0), y(0), z(0) {}
  Pt(dbl x, dbl y, dbl z) : x(x), y(y), z(z) {}
  Pt operator-(const Pt& o)const{return{x-o.x,y-o.y,z-o.z};}
  Pt operator+(const Pt& o)const{return{x+o.x,y+o.y,z+o.z};}
  Pt operator/(const dbl& a) const { return {x/a, y/a, z/a}; }
  Pt operator*(const dbl& a) const { return {x*a, y*a, z*a}; }
  Pt cross(const Pt& o) const {
    dbl nx = y * o.z - z * o.y;
    dbl ny = z * o.x - x * o.z;
    dbl nz = x * o.y - y * o.x;
    return {nx, ny, nz};
  }
  dbl dot(const Pt &o) const { return x*o.x + y*o.y + z*o.z; }
  bool operator==(const Pt& o) const {
    return abs(x-o.x)<EPS && abs(y-o.y)<EPS && abs(z-o.z)<EPS;
  }
  dbl dist() { return sqrtl(x * x + y * y + z * z); }
};

struct Plane {
  dbl a, b, c, d;
  Plane(dbl a_, dbl b_, dbl c_, dbl d_) : a(a_), b(b_), c(c_), d(d_) {
    dbl z = sqrtl(a * a + b * b + c * c);
    if (z < EPS) return;
    a /= z, b /= z, c /= z, d /= z;
  }
  dbl get_val(const Pt &p) const { // НЕ СТАВИТЬ МОДУЛЬ
    return a * p.x + b * p.y + c * p.z + d;
  }
  dbl dist(const Pt &p) const { return abs(get_val(p)); }
  bool on_plane(const Pt &p) const {
    return dist(p) / sqrtl(a * a + b * b + c * c) < EPS;
  }
  Pt proj(const Pt &p) const {
    dbl t = get_val(p) / (a * a + b * b + c * c);
    return p - Pt(a, b, c) * t;
  }
};

bool on_line(Pt p1, Pt p2, Pt p3) {
  return (p2 - p1).cross(p3 - p1) == Pt(0, 0, 0);
}

Plane get_plane(Pt p1, Pt p2, Pt p3) {
  Pt norm = (p2 - p1).cross(p3 - p1);
  Plane pl(norm.x, norm.y, norm.z, 0);
  pl.d = -pl.get_val(p1);
  return pl;
}

// две точки прямой пересечения на расстоянии 1
pair<Pt, Pt> intersect(Plane pl1, Plane pl2) {
  Pt n1(pl1.a, pl1.b, pl1.c), n2(pl2.a, pl2.b, pl2.c);
  Pt dir = n1.cross(n2);
  dbl len2 = dir.dot(dir);
  if (len2 < EPS * EPS) return {Pt(), Pt()}; // параллельны
  Pt p = (n1 * pl2.d - n2 * pl1.d).cross(dir) / len2;
  return {p, p + dir / sqrtl(len2)};
}

// угол между векторами в [0, pi]: сонаправленные -> 0, противоположные -> pi
// угол между прямыми с направляющими p1, p2: min(a, pi - a)
dbl get_ang(Pt p1, Pt p2) {
  return atan2l(p1.cross(p2).dist(), p1.dot(p2));
}

// любой перпендикулярный вектор
Pt vector_perp(Pt v) {
  if (abs(v.x) > EPS || abs(v.y) > EPS)
    return {v.y, -v.x, 0};
  return {v.z, 0, -v.x};
}

// плоскость через точку p перпендикулярная вектору v
Plane plane_perp(Pt p, Pt v) {
  return Plane(v.x, v.y, v.z, -v.dot(p));
}

// ближайшая к p точка прямой ab; при |a-b| < EPS ответ a
Pt closest_line(Pt p, Pt a, Pt b) {
  Pt v = b - a;
  dbl len2 = v.dot(v);
  if (len2 < EPS * EPS) return a;
  return a + v * ((p - a).dot(v) / len2);
}

// ближайшая к p точка луча из a в сторону b
Pt closest_ray(Pt p, Pt a, Pt b) {
  if ((p - a).dot(b - a) < EPS) return a;
  return closest_line(p, a, b);
}

// ближайшая к p точка отрезка ab
Pt closest_seg(Pt p, Pt a, Pt b) {
  if ((p - b).dot(a - b) < EPS) return b;
  return closest_ray(p, a, b);
}

// прямая a + (b - a) * t; |b-a| и длина нормали > EPS
// 0 -- нет, 1 -- точка, 2 -- прямая лежит в плоскости
// t записывается при ответе 1
// луч: t >= 0; отрезок: 0 <= t <= 1
int intersect(Pt a, Pt b, Plane pl, dbl &t) {
  Pt v = b - a, n(pl.a, pl.b, pl.c);
  dbl len = n.dist(), den = n.dot(v), val = pl.get_val(a);
  assert(len > EPS && v.dot(v) > EPS * EPS);
  if (abs(den) < EPS * max<dbl>(1, len * v.dist()))
    return abs(val) < EPS * len ? 2 : 0;
  t = -val / den;
  return 1;
}

// ортонормированный базис плоскости через o; |n| > EPS
struct Basis {
  Pt o, u, v;
  Basis(Pt o, Pt n) : o(o) {
    dbl len = n.dist();
    assert(len > EPS);
    n = n / len;
    u = vector_perp(n);
    u = u / u.dist();
    v = n.cross(u);
  }
  // для точки вне плоскости -- координаты её проекции
  pair<dbl, dbl> to_2d(Pt p) const {
    p = p - o;
    return {p.dot(u), p.dot(v)};
  }
  Pt from_2d(dbl x, dbl y) const { return o + u * x + v * y; }
};

  bool pla = is2plane(a);
  bool plb = is2plane(b);
  if (pla != plb) return pla < plb;
  return sign(a * b) > 0;
}

struct line {

  line(const vctr A, const vctr B) {
    a = A.y - B.y;
    b = B.x - A.x;
    c = A * B;
    // left halfplane of A->B is positive
    // assert(a != 0 || b != 0);
  }
  void operator*=(dbl x) { a *= x, b *= x, c *= x; }
  void operator/=(dbl x) { a /= x, b /= x, c /= x; }
  dbl get(const vctr P) const { return a*P.x + b*P.y + c; }
  // requires dbl vctr
  vctr any_point() const {
    dbl d = a * a + b * b;
    return {-a * c / d, -b * c / d};
  }
  // requires dbl line
  void normalize() {
    dbl d = sqrtl(a * a + b * b);
    a /= d, b /= d, c /= d;
  }
};

bool isparallel(line l1, line l2) {
  return sign(l2.a * l1.b - l2.b * l1.a) == 0;
}

// requires dbl vctr
vctr intersection(const line &l1, const line &l2) {
  dbl z = l2.a * l1.b - l2.b * l1.a;
  dbl x = (l1.c * l2.b - l2.c * l1.b) / z;
  dbl y = -(l1.c * l2.a - l2.c * l1.a) / z;
  return {x, y};
}

// Серединный перпендикуляр (не биссектриса!), get(A) > 0
line bisection(const vctr A, const vctr B) {
  vctr d = (A - B) * 2;
  return line(d.x, d.y, B.dist2() - A.dist2());
}

  circle(dbl x, dbl y, dbl r) : C(x, y), r(r) {}
  circle(vctr C, dbl r) : C(C), r(r) {}
  // requires dbl vctr
  circle(const vctr A, const vctr B) {
    C = (A + B) / 2, r = (A - B).dist() / 2;
  }
  circle(const vctr A, const vctr B, const vctr D) {
    vctr u = B-A, v = D-A;
    dbl p = u.dist2(), q = v.dist2();
    C = A + vctr(v.y*p - u.y*q, u.x*q - v.x*p) / (2*(u*v));
    r = (C - A).dist();
  }
  bool isin(const vctr P, char strict=0) const {
    return sign((C - P).dist() - r) <= -strict;
    // int C, r: exact, but fails on dbl precision
    // return sign((C - P).dist2() - r * r) <= -strict;
  }
};

// below: requires dbl vctr and dbl line
vector<vctr> intersection_line_circ(line l, circle c) {
  l.normalize();
  dbl s = l.get(c.C), d = abs(s);
  if (sign(d - c.r) > 0) return {};
  vctr a = c.C - vctr(l.a, l.b) * s;
  if (sign(c.r - d) == 0) return {a};
  dbl k = sqrtl((c.r - d) * (c.r + d));
  vctr v = vctr(l.b, -l.a) * k;
  // arc from res[0] to res[1] ccw lies in positive halfplane
  return {a + v, a - v};
}

vector<vctr> intersection_circ_circ(circle A, circle B) {
  vctr d = B.C - A.C; // better precision
  line l(2*d.x, 2*d.y, (B.r-A.r) * (B.r+A.r) - d.dist2());
  if (sign(l.a) == 0 && sign(l.b) == 0) return {};
  auto c = A.r<B.r ? circle(0,0,A.r) : circle(d,B.r); // better precision
  auto res = intersection_line_circ(l, c);
  for (auto &p : res) p += A.C;
  // arc from res[0] to res[1] ccw over A is inside B
  // <=> ccw over B is outside A
  return res;
}

vector<vctr> tangent_vctr_circ(vctr v, circle c) {
  dbl d = (c.C - v).dist();
  if (sign(d - c.r) < 0) return {};
  if (sign(d - c.r) == 0) return {v};
  dbl k = sqrtl((d - c.r) * (d + c.r));
  // res[0] is right tangent, res[1] is left tangent
  return intersection_circ_circ(circle(v.x, v.y, k), c);
}

// {point on A, point on B}: outer tangents, then inner.
// 4 apart, 3 touch outside, 2 intersect, 1 touch inside,
// 0 nested. Tangent at touching point has equal points
// (line is perpendicular to A.C->B.C)
// Returns (looking from A.C to B.C):
// {left, left}, {right, right}, {left, right}, {right, left}
vector<pair<vctr, vctr>> common_tangents(circle A, circle B) {
  vctr d = B.C - A.C;
  dbl d2 = d.dist2();
  if (sign(d2) == 0) return {};
  vector<pair<vctr, vctr>> res;
  for (dbl r : {B.r, -B.r}) { // outer, inner
    dbl dr = A.r - r, h2 = d2 - dr * dr;
    if (sign(h2) < 0) continue;
    dbl h = sqrtl(max<dbl>(h2, 0));
    for (int s : {1, -1}) {
      vctr v = (d * dr + d.rotate_ccw_90() * (s * h)) / d2;
      res.pb({A.C + v * A.r, B.C + v * r});
      if (sign(h2) == 0) break;
    }
  }
  return res;
}

// p convex (CCW or CW), no 3 collinear, a outside.
// {right, left} tangents looking from a to p
pair<int, int> tangents_from_point(vctr &a, vector<vctr> &p) {
  int n = p.size();
  int logn = 31 - __builtin_clz(n);
  auto find_with_sign = [&](int sgn) {
    int i = 0;
    for (int k = logn; k >= 0; --k) {
      int i1 = (i - (1 << k) + n) % n;
      int i2 = (i + (1 << k)) % n;
      if (sign((p[i1] - a) * (p[i] - a)) == sgn) i = i1;
      if (sign((p[i2] - a) * (p[i] - a)) == sgn) i = i2;
    }
    return i;
  };
  return {find_with_sign(1), find_with_sign(-1)};
}

// min (sgn=-1) signed distance to line

    int i1 = (i - (1 << k) + n) % n;
    int i2 = (i + (1 << k)) % n;
    if (l.get(p[i1]) > l.get(p[i])) i = i1;
    if (l.get(p[i2]) > l.get(p[i])) i = i2;
  }
  return i;
}

                             bool strict = 0) {

    if (!sign(u * v) && sign(u % v) <= 0) return !strict;

// For int line let $N=\max(|a|,|b|)$, $K=\max|c|$, then:
// $O(\log)$ `compare` needs $2N^2$ and $2NK$ fit into type;
// cross-multiplication `compare` needs $4N^3K$.

// Half plane: ax+by+c >= 0. Bounding box MUST HAVE.
vector<int> calc_hpi_inds(const vector<line> &ls) {
  int n = ls.size(), m=0, l=0, r=0;
  vector<int> q(n);
  vector<char> h(n);
  iota(all(q), 0);
  auto cr = [&](int i, int j) {
    return ls[i].a*ls[j].b-ls[i].b*ls[j].a;
  };
  auto bigb = [&](int i){return abs(ls[i].b)>=abs(ls[i].a);};
  auto num = [&](int i, int j) {
    if (bigb(j)) return ls[i].b*ls[j].c-ls[i].c*ls[j].b;
    return ls[i].c*ls[j].a-ls[i].a*ls[j].c;
  };
  auto dir = [&](int i) {
    return bigb(i) ? sign(ls[i].b) : -sign(ls[i].a);
  };
  auto on = [&](int i, int j){return -sign(num(i,j))*dir(j);};
  auto at = [&](int i, int j, int k) {
    // return (int)sign(ls[k].get(intersection(ls[i],ls[j])));
    // uncomment above and comment code below if float line
    if (!cr(j,k)) return on(k,j);
    return -sign(cr(j,k))*dir(j)*compare(
      num(i,j),cr(i,j),num(k,j),cr(k,j));
  };
  auto same_dir = [&](int i, int j) {
    return !sign(cr(i,j))&&ls[i].a*ls[j].a+ls[i].b*ls[j].b>0;
  };
  for (int i = 0; i < n; ++i)
    h[i] = sign(ls[i].b)<0||(!sign(ls[i].b)&&ls[i].a<0);
  sort(all(q), [&](int i, int j) { // 70% of time
    return h[i] != h[j] ? h[i] < h[j] : cr(i,j) > 0;
  });
  for (int i : q) {
    if (!m || !same_dir(q[m-1],i)) q[m++] = i;
    else if (on(q[m-1],i) >= 0) q[m-1] = i;
  }

  auto bad = [&](int i, int j, int k) {
    if (!sign(cr(i,j))) return false;
    int s = at(i,j,k), t = sign(cr(i,k));
    return s<0 || (!s&&t&&sign(cr(i,j))==t&&sign(cr(j,k))==t);
  };
  for (int t = 0; t < m; ++t) {
    int i = q[t];
    while (r-l>1 && bad(q[r-2],q[r-1],i)) --r;
    while (r-l>1 && bad(q[l+1],q[l],i)) ++l;
    if (l<r && same_dir(q[r-1],i) && on(q[r-1],i)<0)
      return {};
    q[r++] = i;
  }
  while (r-l>2 && bad(q[r-2],q[r-1],q[l])) --r;
  while (r-l>2 && bad(q[l+1],q[l],q[r-1])) ++l;
  if (r-l < 3) return {};
  return {q.begin()+l,q.begin()+r};
}
// requires dbl vctr (line can be integer)
vector<vctr> half_plane_intersection(const vector<line> &ls) {
  vector<int> inds = calc_hpi_inds(ls);
  int n = inds.size();
  vector<vctr> pts; pts.reserve(n);
  for (int i = 0; i < n; ++i) {
    vctr P = intersection(ls[inds[i]],ls[inds[(i+1)%n]]);
    if (pts.empty() || !(pts.back() == P)) pts.pb(P);
  }
  if (pts.size()>1 && pts[0] == pts.back()) pts.pop_back();
// "empty"->{}, "point"->{P}, "segment"->{A,B}, else CCW.
// `inds` have no lines with 0 length, except for "segment"
// (4 lines), and "point" (3 lines, or 4 if 2 and 2 parallel)
  return pts;
}

struct TwoSat {
  int n;
  vector<vector<int>> g, rg;
  vector<int> comp, topsort;
  vector<char> used;

  TwoSat(int n) : n(n), g(2 * n), rg(2 * n) {
    topsort.reserve(2 * n);
  }

  int neg(int v) { return 2 * n - 1 - v; }

  void add(int v, int u) {
    g[v].pb(u);
    rg[u].pb(v);
  }

  void add_OR(int v, int u) { // v | u
    add(neg(v), u), add(neg(u), v);
  }

  void add_IMPL(int v, int u) { add_OR(neg(v), u); } // v -> u

  void dfs1(int v) {
    used[v] = 1;
    for (auto u: g[v])
      if (!used[u]) dfs1(u);
    topsort.pb(v);
  }

  void dfs2(int v, int col) {
    comp[v] = col;
    for (auto u: rg[v])
      if (comp[u] == -1) dfs2(u, col);
  }

  void SCC() {
    used.assign(2 * n, 0), comp.assign(2 * n, -1);
    topsort.clear();
    for (int v = 0; v < 2 * n; ++v)
      if (!used[v]) dfs1(v);
    reverse(all(topsort));
    int cc = 0;
    for (int i = 0; i < 2 * n; ++i)
      if (comp[topsort[i]] == -1)
        dfs2(topsort[i], cc++);
  }

  vector<char> solution() {
    assert(n > 0); // returns {} if ans exists AND not
    SCC();
    vector<char> ans(n);
    for (int v = 0; v < n; ++v) {
      if (comp[v] == comp[neg(v)])
        return {}; // no solution
      ans[v] = comp[v] > comp[neg(v)];
    }
    return ans;
  }
};

struct LRFlow {
  Dinic dinic; // ориентированные рёбра
  int S, T; // исток и сток
  int Sx, Tx; // вспомогательные вершины, любые неиспользуемые индексы
  vector<ll> dem;

  LRFlow(int n, int S, int T, int Sx, int Tx)
    : dinic(n), S(S), T(T), Sx(Sx), Tx(Tx), dem(n) {}

  // returns id: real flow = dinic.g[v][id].f + mincap
  int addedge(int v, int u, flow_t mincap, flow_t maxcap) {
    dem[v] -= mincap, dem[u] += mincap;
    return dinic.addedge(v, u, maxcap - mincap);
  }

  // returns true if found flow is feasible.
  // size of flow (after aux edges are removed below):
  // sum e.f over dinic.g[S] - dem[S]
  bool run() {
    ll totaldem = 0;
    for (int v = 0; v < dem.size(); ++v) {
      if (dem[v] > 0) {
        totaldem += dem[v];
        // aux capacities are sums: flow_t must fit totaldem (~ sum of mincaps)
        dinic.addedge(Sx, v, dem[v]);
      } else if (dem[v] < 0)
        dinic.addedge(v, Tx, -dem[v]);
    }
    dinic.addedge(T, S, totaldem); // totaldem is enough
    dinic.addedge(S, T, totaldem); // for total flow < 0
    dinic.run(Sx, Tx);
    ll flow = 0;
    for (auto &e : dinic.g[Sx]) flow += e.f;
    if (flow != totaldem) return false;
    // return true; // if want ONLY feasibility
    for (int v = 0; v < dem.size(); ++v)
      if (dem[v]) dinic.g[v].pop_back();

// Вершины нумераются от 1 до n. Петли и кратные рёбра - ок.
// Ищет паросочетание max веса, число рёбер любое.
namespace weighted_matching{
  const ll INF = 1e18; // > 4 * max w
  const int MAXN = 1050; //double of possible N
  struct E { int x, y; ll w; };
  int n, m;
  E G[MAXN][MAXN];
  ll lab[MAXN];
  int match[MAXN], slack[MAXN], st[MAXN], pa[MAXN];
  int flo_from[MAXN][MAXN], S[MAXN], vis[MAXN];
  vector<int> flo[MAXN];

    n = _n;

      for(int y = 1; y <= n; ++y)
        G[x][y] = E{x, y, 0};
  }
  void add_edge(int x, int y, ll w) {
    G[x][y].w = G[y][x].w = max(G[x][y].w, w);
  }
  ll e_delta(E e) {
    return lab[e.x] + lab[e.y] - G[e.x][e.y].w * 2;
  }
  void update_slack(int u, int x) {
    if(!slack[x] || e_delta(G[u][x]) < e_delta(G[slack[x]][x]))
      slack[x] = u;
  }
  void set_slack(int x) {
    slack[x] = 0;
    for(int u = 1; u <= n; ++u)
      if(G[u][x].w > 0 && st[u] != x && S[st[u]] == 0)
        update_slack(u, x);

    if(x <= n) Q.push(x);
    else for(int y : flo[x]) q_push(y);

    st[x] = b;

      for(int y : flo[x]) set_st(y, b);

    int pr = find(all(flo[b]), xr) - flo[b].begin();
    if(pr & 1) {
      reverse(flo[b].begin() + 1, flo[b].end());
      return ssize(flo[b]) - pr;
    } else return pr;
  }
  void set_match(int x, int y) {
    match[x] = G[x][y].y;
    if(x <= n) return;
    E e = G[x][y];
    int xr = flo_from[x][e.x], pr = get_pr(x, xr);
    for(int i = 0; i < pr; ++i) set_match(flo[x][i], flo[x][i^1]);
    set_match(xr, y);
    rotate(flo[x].begin(), flo[x].begin() + pr, flo[x].end());
  }
  void augment(int x, int y) {
    while(1) {
      int ny = st[match[x]];
      set_match(x, y);
      if(!ny) return;
      set_match(ny, st[pa[ny]]);
      x = st[pa[ny]], y = ny;
    }
  }

    static int t = 0;
    for(++t; x || y; swap(x, y)) {
      if(x == 0) continue;
      if(vis[x] == t) return x;
      vis[x] = t, x = st[match[x]];
      if(x) x = st[pa[x]];
    }
    return 0;
  }
  void add_blossom(int x, int l, int y) {
    int b = n + 1;
    while(b <= m && st[b]) ++b;
    if(b > m) ++m;
    lab[b] = 0, S[b] = 0, match[b] = match[l];
    flo[b].clear(), flo[b].pb(l);
    for(int u = x, v; u != l; u = st[pa[v]])
      flo[b].pb(u), flo[b].pb(v = st[match[u]]), q_push(v);
    reverse(flo[b].begin() + 1, flo[b].end());
    for(int u = y, v; u != l; u = st[pa[v]])
      flo[b].pb(u), flo[b].pb(v = st[match[u]]), q_push(v);
    set_st(b, b);
    for(int i = 1; i <= m; ++i) G[b][i].w = G[i][b].w = 0;
    for(int i = 1; i <= n; ++i) flo_from[b][i] = 0;
    for(int us : flo[b]) {
      for(int u = 1; u <= m; ++u)
        if(G[b][u].w == 0 || e_delta(G[us][u]) < e_delta(G[b][u]))
          G[b][u] = G[us][u], G[u][b] = G[u][us];
      for(int u = 1; u <= n; ++u)
        if(flo_from[us][u])
          flo_from[b][u] = us;
    }
    set_slack(b);
  }
  void expand_blossom(int b) {
    for(int x : flo[b]) set_st(x, x);
    int xr = flo_from[b][G[b][pa[b]].x], pr = get_pr(b, xr);
    for(int i = 0; i < pr; i += 2) {
      int xs = flo[b][i], xns = flo[b][i + 1];
      pa[xs] = G[xns][xs].x;
      S[xs] = 1, S[xns] = 0;
      slack[xs] = 0, set_slack(xns);
      q_push(xns);
    }
    S[xr] = 1, pa[xr] = pa[b];
    for(int i = pr + 1; i < flo[b].size(); ++i) {
      int xs = flo[b][i];
      S[xs] = -1, set_slack(xs);
    }
    st[b] = 0;
  }
  bool on_found_edge(E e) {
    int x = st[e.x], y = st[e.y];
    if(S[y] == -1) {
      pa[y] = e.x, S[y] = 1;
      int ny = st[match[y]];
      slack[y] = slack[ny] = 0;
      S[ny] = 0, q_push(ny);
    } else if(S[y] == 0) {
      int l = get_lca(x, y);
      if(!l) return augment(x, y), augment(y, x), true;
      else add_blossom(x, l, y);
    }
    return false;
  }
  bool matching() {
    fill(S + 1, S + m + 1, -1);
    fill(slack + 1, slack + m + 1, 0);
    Q = queue<int>();
    for(int x = 1; x <= m; ++x)
      if(st[x] == x && !match[x]) pa[x] = 0, S[x] = 0, q_push(x);
    if(Q.empty()) return false;
    while(1) {
      while(Q.size()) {
        int x = Q.front(); Q.pop();
        if(S[st[x]] == 1) continue;
        for(int y = 1; y <= n; ++y) {
          if(G[x][y].w > 0 && st[x] != st[y]) {
            if(e_delta(G[x][y]) == 0) {
              if(on_found_edge(G[x][y])) return true;
            } else update_slack(x, st[y]);
          }
        }
      }
      ll d = INF;
      for(int b = n + 1; b <= m; ++b)
        if(st[b] == b && S[b] == 1) d = min(d, lab[b] / 2);
      for(int x = 1; x <= m; ++x)
        if(st[x] == x && slack[x]) {
          if(S[x] == -1) d = min(d, e_delta(G[slack[x]][x]));
          else if(S[x] == 0) d = min(d, e_delta(G[slack[x]][x]) / 2);
        }
      for(int x = 1; x <= n; ++x) {
        if(S[st[x]] == 0) {
          if(lab[x] <= d) return 0;
          lab[x] -= d;
        } else if(S[st[x]] == 1) lab[x] += d;
      }
      for(int b = n + 1; b <= m; ++b)
        if(st[b] == b) {
          if(S[st[b]] == 0) lab[b] += d * 2;
          else if(S[st[b]] == 1) lab[b] -= d * 2;
        }
      Q = queue<int>();
      for(int x = 1; x <= m; ++x)
        if(st[x] == x && slack[x] && st[slack[x]] != x && e_delta(G[slack[x]][x]) == 0)
          if(on_found_edge(G[slack[x]][x])) return true;
      for(int b = n + 1; b <= m; ++b)
        if(st[b] == b && S[b] == 1 && lab[b] == 0)
          expand_blossom(b);
    }
    return false;
  }
  pair<ll, int> solve(vector<pair<int, int>> &ans) {
    fill(match + 1, match + n + 1, 0);
    m = n;
    int cnt = 0; ll sum = 0, mx = 0;
    for(int u = 0; u <= n; ++u) st[u] = u, flo[u].clear();
    for(int x = 1; x <= n; ++x)
      for(int y = 1; y <= n; ++y){
        flo_from[x][y] = (x == y ? x : 0);
        mx = max(mx, G[x][y].w);
      }
    for(int x = 1; x <= n; ++x) lab[x] = mx;
    while(matching()) ++cnt;
    for(int x = 1; x <= n; ++x)
      if(match[x] && match[x] < x) {
        sum += G[x][match[x]].w;
        ans.pb({x, G[x][match[x]].y});
      }
    return {sum, cnt};
  }
}

pair<ll, vector<int>> venger(vector<vector<ll>> a) {
// ищет минимальное по стоимости
// работает только при n <= m
// a - массив весов $(n+1) \times (m+1)$
// a[0][..] = a[..][0] = 0
// a[i][j] любые (>0, <0, =0), n * max|a[i][j]| <= 1.5e18
// возвращает ans[i] = j если взяли ребро a[i][j]
  int n = ssize(a) - 1;
  int m = ssize(a[0]) - 1;
  vector<ll> u(n + 1), v(m + 1);
  vector<int> p(m + 1), way(m + 1);
  const ll INF = LLONG_MAX;
  for (int i = 1; i <= n; ++i) {
    p[0] = i;
    int y = 0;
    vector<ll> minv(m + 1, INF);
    vector<char> used(m + 1, false);
    do {
      used[y] = true;
      int x = p[y], z;
      ll delta = INF;
      for (int j = 1; j <= m; ++j) {
        if (!used[j]) {
          ll cur = a[x][j] - u[x] - v[j];
          if (cur < minv[j])
            minv[j] = cur, way[j] = y;
          if (minv[j] < delta)
            delta = minv[j], z = j;
        }
      }
      for (int j = 0; j <= m; ++j) {
        if (used[j])
          u[p[j]] += delta, v[j] -= delta;
        else
          minv[j] -= delta;
      }
      y = z;
    } while (p[y] != 0);
    do {
      int z = way[y];
      p[y] = p[z], y = z;
    } while (y);
  }
  ll cost = -v[0];
  vector<int> ans(n + 1);
  for (int j = 1; j <= m; ++j) ans[p[j]] = j;
  return {cost, ans};
}

// flow_t: capacity of one edge, total flow is ll
// undirected edges: flow_t must fit 2c
using flow_t = int;
const int LOG = 29; // масштабирование, =0 если не нужно
struct Edge { int u, r; flow_t f, c; };

struct Dinic {
  vector<vector<Edge>> g;
  vector<char> vis;
  vector<int> inds, dist, Q;
  int ql, qr, S, T;
  flow_t BIT; // 2^LOG must fit flow_t
  Dinic(int n) : g(n), vis(n), inds(n), dist(n), Q(n) {}

  bool bfs() {
    memset(vis.data(), 0, vis.size());
    ql = 0, qr = 0;
    dist[S] = 0, vis[S] = true, Q[qr++] = S;
    while (ql < qr) {
      int v = Q[ql++];
      for (auto &e : g[v]) {
        int u = e.u;
        if (vis[u] || e.c - e.f < BIT)
          continue;
        vis[u] = true, dist[u] = dist[v] + 1;
        Q[qr++] = u;
        if (u == T) return true;
      }
    }
    return false;
  }

  flow_t dfs(int v, flow_t maxF) {
    if (v == T) return maxF;
    flow_t ans = 0;
    for (int &i = inds[v]; i < g[v].size(); ++i) {
      auto &e = g[v][i];
      auto cc = min(maxF - ans, e.c - e.f);
      if (dist[e.u] <= dist[v] || !vis[e.u] ||
          inds[e.u] == g[e.u].size() || cc < BIT)
        continue;
      auto f = dfs(e.u, cc);
      if (f != 0) {
        e.f += f, ans += f;
        g[e.u][e.r].f -= f;
      }
      // иногда быстрее один иф, иногда другой
      if (maxF - ans < 1) break;
      // if (maxF - ans < BIT) break;
    }
    return ans;
  }

  // repeated runs are NOT independent
  void run(int s, int t) {
    S = s, T = t;
    assert(S != T);
    for (BIT = (1ll << LOG); BIT > 0; BIT >>= 1) {
      while (bfs()) {
        memset(inds.data(), 0, inds.size() * sizeof(int));
        for (auto &e : g[S]) {
          if (inds[e.u] == g[e.u].size() || e.c - e.f < BIT)
            continue;
          flow_t f = dfs(e.u, e.c - e.f);
          e.f += f, g[e.u][e.r].f -= f;
        }
      }
    }
  }

  int addedge(int v, int u, flow_t c) { // index in g[v]
    int i = g[v].size();
    g[v].pb({u, (int)g[u].size(), 0, c});
    // если не ориентированно, то обратная capacity = c
    g[u].pb({v, i, 0, 0});
    return i;
  }
};

\subsection{Минкост}
\begin{lstlisting}[language=C++]
// flow_t: capacity of one edge, total flow is ll
// cost_t: must fit n * max|w| and the answer
using flow_t = int;
using cost_t = ll;
const cost_t INFw = 1e18;
const flow_t INFf = 2e9;

struct MinCost {
  struct Edge { int v, u; flow_t f, c; cost_t w; };
  vector<Edge> edg; // edg[i ^ 1] is reverse of edg[i]
  vector<vector<int>> g;
  vector<cost_t> dist, pot;
  vector<int> pre;
  vector<char> inQ;
  int S, T;
  MinCost(int n) : g(n), dist(n), pot(n), pre(n), inQ(n) {}

  flow_t get_flow() {
    int v = T;
    if (pre[v] == -1) return 0;
    flow_t f = INFf;
    do {
      int ei = pre[v];
      Edge &e = edg[ei];
      f = min(f, e.c - e.f);
      if (f == 0) return 0;
      v = e.v;
    } while (v != S);
    v = T;
    do {
      int ei = pre[v];
      edg[ei].f += f;
      edg[ei ^ 1].f -= f;
      v = edg[ei].v;
    } while (v != S);
    return f;
  }

  void spfa() {
    fill(all(dist), INFw);
    fill(all(pre), -1);
    dist[S] = 0;
    deque<int> Q = {S};
    inQ[S] = true;
    while (!Q.empty()) {
      int v = Q.front();
      Q.pop_front(), inQ[v] = false;
      cost_t d = dist[v];
      for (int ei : g[v]) {
        Edge &e = edg[ei];
        if (e.f == e.c) continue;
        cost_t w = e.w + pot[v] - pot[e.u];
        if (dist[e.u] <= d + w) continue;
        pre[e.u] = ei, dist[e.u] = d + w;
        if (!inQ[e.u])
          inQ[e.u] = true, Q.pb(e.u);
      }
    }
    for (int i = 0; i < dist.size(); ++i)
      if (dist[i] < INFw) // else pot overflows
        pot[i] += dist[i];
  }

  // returns {flow, cost}
  // repeated runs are NOT independent
  pair<ll, cost_t> run(int s, int t) {
    S = s, T = t;
    spfa(); // pot[i] = 0 // or ford_bellman
    ll flow = 0;
    while (true) {
      flow_t ff = get_flow();
      if (ff == 0) break;
      flow += ff;
      spfa(); // or dijkstra
    }
    cost_t cost = 0;
    for (int i = 0; i < edg.size(); i += 2)
      cost += edg[i].f * edg[i].w;
    return {flow, cost};
  }

  // returns id: flow on the edge = edg[id].f
  int addedge(int v, int u, flow_t c, cost_t w) {
    // reserve edg to m*2 before adding
    int i = edg.size();
    edg.pb({v, u, 0, c, w});
    edg.pb({u, v, 0, 0, -w});
    g[v].pb(i);
    g[u].pb(i + 1);
    return i;
  }
};

void use_example() {
  MinCost mc(n);
  for (int i = 0; i < m; ++i) {
    int v, u; flow_t c; cost_t w;

// s==t => непустой цикл (даже при |need|=0).
// O(2^|need| (n+m) ans)
ull path_mul(ull a, ull b) {
  ull r = 0;
  for (; b; b >>= 1, a = (a << 1) ^ (a >> 63) * 27)
    r ^= a * (b & 1);
  return r;
}

// Возвращает {число рёбер, первая вершина после s}, либо {-1,-1} если нет.
pair<int, int> find_path_through_specified(int n,
    vector<pair<int, int>> edges, int s, int t,
    vector<int> need) {
  vector<int> bit(n);
  int k = 0;
  for (auto v : need) {
    if (v == s || v == t || bit[v]) continue;
    bit[v] = 1 << k++;
  }
  int masks = 1 << k;
  struct Edge { int v, u; ull w; };
  vector<Edge> es;
  vector<int> from, to;
  for (auto [v, u] : edges) {
    if ((v == s && u == t) || (v == t && u == s)) {
      if (k == 0) return {1, t};
      continue;
    }
    if (v == s || u == s) from.pb(v ^ u ^ s);
    if (v == t || u == t) to.pb(v ^ u ^ t);
    if (v != s && v != t && u != s && u != t) {
      ull w = rnd();
      es.pb({v, u, w});
      es.pb({u, v, w});
    }
  }
  int m = es.size();
  vector<vector<ull>> dp(masks, vector<ull>(n));
  vector<vector<ull>> dp_e(masks, vector<ull>(m));
  auto ndp = dp, ndp_e = dp_e;
  for (auto v : to) dp[bit[v]][v] = rnd();
  for (int len = 1; len < n; ++len) {
    if (s != t || len >= 2) {
      for (auto v : from)
        if (dp[masks - 1][v]) return {len + 1, v};
    }
    if (len + 1 == n) break;
    for (auto &r : ndp) fill(all(r), 0);
    for (auto &r : ndp_e) fill(all(r), 0);
    for (int mask = 0; mask < masks; ++mask) {
      for (int id = 0; id < m; ++id) {
        auto [v, u, w] = es[id];
        if (bit[u] && (mask & bit[u])) continue;
        int nmask = mask | bit[u];
        ull x = dp[mask][v] ^ dp_e[mask][id ^ 1];
        if (!x) continue;
        x = path_mul(x, w);
        ndp[nmask][u] ^= x;
        ndp_e[nmask][id] ^= x;
      }
    }
    swap(dp, ndp);
    swap(dp_e, ndp_e);
  }
  return {-1, -1};
}

    t[v].l = add_on_seg(t[v].l, l, m, ql, qr, ln);

void unlimited_knapsack(int n, int T, vector<int> &v) {
  // O(n^2 log n + T)
  // v[1..n], v[i] --- best value of a single item with weight=i (-inf for non-existent)
  vector<int> items;
  for (int i = 1; i <= n; ++i)
    if (v[i] > -inf) items.pb(i);
  sort(all(items), [&](int i, int j) {
    return v[i] * j > v[j] * i;
  });
  v.resize(T + 1, -inf), v[0] = 0;
  for (int i = 2; i <= T; ++i) {
    int tick = 0;
    for (auto j: items) {
      if (i > j)
        v[i] = max(v[i], v[j] + v[i - j]);
      if (tick * i > 3 * n * n) break; // apparently 2 works as well, but proved bound is 3
      tick++;
    }
  }
  // v[i] --- best total value of items with total weight=i
  // weight i is unreachable if `v[i] < -inf/2` (not `== -inf`)
}

struct Node {
  Node *ch[2] = {}, *p = nullptr;
  bool rev = false;
  int sz = 1;
};

int size(Node *v) { return (v ? v->sz : 0); }

int chnum(Node *v) { return v->p->ch[1] == v; }

bool isroot(Node *v) {
  return v->p == nullptr || v->p->ch[chnum(v)] != v;
}

void push(Node *v) {
  if (v->rev) {
    if (v->ch[0]) v->ch[0]->rev ^= 1;
    if (v->ch[1]) v->ch[1]->rev ^= 1;
    swap(v->ch[0], v->ch[1]);
    v->rev = false;
  }
}

void pull(Node *v) {
  v->sz = size(v->ch[1]) + size(v->ch[0]) + 1;
}

void attach(Node *v, Node *p, int num) {
  if (p) p->ch[num] = v;
  if (v) v->p = p;
}

void rotate(Node *v) {
  Node *p = v->p;
  push(p);
  push(v);
  int num = chnum(v);
  Node *u = v->ch[1 - num];
  if (!isroot(v->p)) attach(v, p->p, chnum(p));
  else v->p = p->p;
  attach(u, p, num);
  attach(p, v, 1 - num);
  pull(p);
  pull(v);
}

void splay(Node *v) {
  push(v);
  while (!isroot(v)) {
    if (!isroot(v->p)) {
      if (chnum(v) == chnum(v->p))
        rotate(v->p);
      else
        rotate(v);
    }
    rotate(v);
  }
}

void expose(Node *v) {
  splay(v), v->ch[1] = nullptr, pull(v);
  while (v->p != nullptr) {
    Node *p = v->p;
    splay(p), attach(v, p, 1), pull(p), splay(v);
  }
}

void makeroot(Node *v) {
  expose(v), v->rev ^= 1, push(v);
}

void link(Node *v, Node *u) {
  makeroot(v);
  makeroot(u);
  u->p = v;
}

void cut(Node *v, Node *u) {
  makeroot(u);
  makeroot(v);
  v->ch[1] = nullptr;
  u->p = nullptr;
  pull(v);
}

int get(Node *v, Node *u) {
  makeroot(u);
  makeroot(v);
  Node *w = u;
  while (!isroot(w)) w = w->p;
  return (w == v ? size(v) - 1 : -1);

// __builtin_ctz = Count Trailing Zeroes
// __builtin_clz = Count Leading Zeroes
// both are UB in gcc when pass 0
struct custom_bitset {
  vector<uint64_t> bits;
  int b, n;

  custom_bitset(int bb = 0) { init(bb); }
  void init(int bb) {
    b = bb, n = (b+63)/64;
    bits.assign(n, 0);
  }
  void clear() { b = n = 0, bits.clear(); }
  void reset() { bits.assign(n, 0); }
  void _clean() { // Reset all bits after `b`.
    if (b%64)
      bits.back() &= (1LLU << (b%64)) - 1;
  }
  bool get(int i) const { return bits[i/64] >> (i%64) & 1; }
  void set(int i, bool value) { // assert(0 <= i && i < b);
    bits[i/64] &= ~(1LLU << (i%64));
    bits[i/64] |= uint64_t(value) << (i%64);
  }

  // Simulates `bs1 |= bs2 << s;`
  // `|=` can be replaced with `^=`, `&=`, `=`
  void or_shift_left(int s){return or_shift_left(*this,s);}
  void or_shift_left(const custom_bitset &o, int s) {
    // assert(s >= 0);
    int d = s/64, r = s%64;
    if (r == 0) {
      for (int i = n-1; i >= d; --i)
        bits[i] |= o.bits[i-d];
    } else {
      for (int i = n-1; i >= d+1; --i)
        bits[i] |= o.bits[i-d-1]>>(64-r) | o.bits[i-d]<<r;
      if (d < n) bits[d] |= o.bits[0]<<r;
    }
    // if `&=`, `=`
    //fill(bits.begin(), bits.begin() + min(d, n), 0);
    _clean();
  }

  // Simulates `bs1 |= bs2 >> s;`
  // `|=` can be replaced with `^=`, `&=`, `=`
  void or_shift_right(int s){return or_shift_right(*this,s);}
  void or_shift_right(const custom_bitset &o, int s) {
    // assert(s >= 0);
    int d = s/64, r = s%64;
    if (r == 0) {
      for (int i = d; i < n; ++i)
        bits[i-d] |= o.bits[i];
    } else {
      for (int i = 0; i < n-d-1; ++i)
        bits[i] |= o.bits[i+d+1]<<(64-r) | o.bits[i+d]>>r;
      if (d < n) bits[n-d-1] |= o.bits[n-1]>>r;
    }
    // if `&=`, `=`
    //fill(bits.end() - min(d, n), bits.end(), 0);
    _clean();
  }

  // find min j, that j >= i and bs[j] = 1;
  int find_next(int i) {
    i = max(i,0);
    if (i >= b) return b;
    int d = i/64, r = i%64;
    auto x = bits[d]>>r;
    if (x != 0) return i + __builtin_ctzll(x);
    for (auto k = d+1; k < n; ++k)
      if (bits[k] != 0)
        return 64*k + __builtin_ctzll(bits[k]);
    return b;
  }

  // `|=` can be replaced with `&=`, `^=`
  custom_bitset &operator|=(const custom_bitset &other){
    // assert(b == other.b);
    for (int i = 0; i < n; ++i)
      bits[i] |= other.bits[i];
    return *this;
  }
};

int getkfps(vector<ll> p, vector<ll> q, ll k) {
  // assert(q[0] != 0);
  while (k) {
    auto f = q;
    for (int i = 1; i < f.size(); i += 2)
      f[i] = (MOD - f[i] % MOD) % MOD;
    auto p2 = conv_mod(p, f);
    auto q2 = conv_mod(q, f);
    p.clear(), q.clear();
    for (int i = k % 2; i < p2.size(); i += 2)
      p.pb(p2[i]);
    for (int i = 0; i < q2.size(); i += 2)
      q.pb(q2[i]);
    k >>= 1;
  }
  return p.empty() ? 0 : (p[0] * inverse(q[0])) % MOD;
}

// a - initials values of sequence, s - result of berlekamp on a
int kth_term(vector<ll> &a, vector<ll> s, ll k) {
  int d = ssize(s) - 1;
  s[0] = MOD - 1;
  while (s.back() == 0) s.pop_back();
  for (auto &el: s)
    el = (MOD - el % MOD) % MOD;
  vector<ll> p(d);
  copy(a.begin(), a.begin() + d, p.begin());
  p = conv_mod(p, s);
  p.resize(d);
  return getkfps(p, s, k);
}

vector<ll> berlekamp_massey(vector<ll> a) {
  // given a[0..n], returns s[1..k]: $a_i = \sum_{j=1}^k a_{i-j} s_j$
  vector<ll> ls, s;
  int lf = 0, d = 0;
  for (int i = 0; i < a.size(); ++i) {
    ll t = 0;
    for (int j = 0; j < s.size(); ++j)
      t = (t + 1ll * a[i - j - 1] * s[j]) % MOD;
    if ((t - a[i]) % MOD == 0) continue;
    if (s.empty()) {
      s.resize(i + 1);
      lf = i, d = (t - a[i]) % MOD;
      continue;
    }
    ll k = -(a[i] - t) * inverse(d) % MOD;
    vector<ll> c(i - lf - 1);
    c.pb(k);
    for (auto &j: ls) c.pb(-j * k % MOD);
    if (c.size() < s.size())
      c.resize(s.size());
    for (int j = 0; j < s.size(); ++j)
      c[j] = (c[j] + s[j]) % MOD;
    if (i - lf + ssize(ls) >= ssize(s))
      tie(ls, lf, d) = make_tuple(s, i, (t - a[i]) % MOD);
    s = c;
  }
  s.insert(s.begin(), 0); // fictive s[0] = 0
  for (auto &i: s) i = (i % MOD + MOD) % MOD;
  return s;
}

const int LOG = 20, MAXN = 1 << LOG;

struct comp {
  double x, y; // not dbl: long double is ~3x slower
  comp() : x(0), y(0) {}
  comp(double x, double y) : x(x), y(y) {}
  comp(int x) : x(x), y(0) {}
  comp operator+(const comp &o) const { return {x + o.x, y + o.y}; }
  comp operator-(const comp &o) const { return {x - o.x, y - o.y}; }
  comp operator*(const comp &o) const { return {x * o.x - y * o.y, x * o.y + y * o.x}; }
  comp operator/(const int k) const { return {x / k, y / k}; }
  comp conj() const { return {x, -y}; }
};

comp OMEGA[MAXN + 10];
int tail[MAXN + 10];
bool fftinit = false;

comp omega(int n, int k) { return OMEGA[MAXN / n * k]; }

int gettail(int x, int lg) { return tail[x] >> (LOG - lg); }

void calcomega() {
  for (int i = 0; i < MAXN; ++i) {
    double x = 2 * PI * i / MAXN;
    OMEGA[i] = {cos(x), sin(x)};
  }
}

void calctail() {
  tail[0] = 0;
  for (int i = 1; i < MAXN; ++i)
    tail[i] = (tail[i/2]/2) | ((i & 1) << (LOG - 1));
}

void fft(vector<comp> &A, int lg) {
  if (!fftinit) calcomega(), calctail(), fftinit = 1;
  int n = A.size();
  for (int i = 0; i < n; ++i)
    if (int j = gettail(i, lg); i < j)
      swap(A[i], A[j]);
  for (int len = 2; len <= n; len *= 2) {
    for (int i = 0; i < n; i += len) {
      for (int j = 0; j < len / 2; ++j) {
        auto v = A[i + j];
        auto u = A[i + j + len / 2] * omega(len, j);
        A[i + j] = v + u;
        A[i + j + len / 2] = v - u;
      }
    }
  }
}

// optional, can be skipped
void fft2(vector<comp> &A, vector<comp> &B, int lg) {
  int n = A.size();
  vector<comp> C(n);
  for (int i = 0; i < n; ++i) {
    C[i].x = A[i].x;
    C[i].y = B[i].x;
  }
  fft(C, lg);
  C.pb(C[0]);
  for (int i = 0; i < n; ++i) {
    A[i] = (C[i] + C[n - i].conj()) / 2;
    B[i] = (C[i] - C[n - i].conj()) / 2 * comp(0, -1);
  }
}

  fft(A, lg);
  for (auto &el : A) el = el / n;
  reverse(A.begin() + 1, A.end());
}

\subsection{Floor Sum, Min Rem}
\begin{lstlisting}[language=C++]
// f(x) = (a*x+b) mod m in [0,m); ll = __int128.
// S = {0} U {x > 0 : f(x) < min_{0 <= j < x} f(j)}.
// {X,DX}: S = {0} U {X[i] + t*DX[i] : 0 <= i < DX.size(),
//   1 <= t <= (X[i+1]-X[i])/DX[i]}; X[0] = 0; O(log m).
pair<vector<ll>, vector<ll>> min_of_linear_segments(ll a, ll b, ll m) {
  a = (a % m + m) % m, b = (b % m + m) % m;
  ll g = __gcd(a, m);
  ll A = a / g, M = m / g, y = b / g;
  ll q = 1, s = 1, dl = M - A, dr = A, x = 0;
  vector<ll> X = {0}, DX;
  while (y) {
    ll k = dr / dl;
    dr %= dl;
    if (!dr) --k, dr = dl;
    s += k * q;
    while (true) {
      k = max<ll>(0, (dl - y + dr - 1) / dr);
      if (dl - k * dr <= 0) break;
      dl -= k * dr;
      q += k * s;
      k = y / dl;
      y -= k * dl, x += q * k;
      X.pb(x), DX.pb(q);
    }
    k = dl / dr;
    dl -= k * dr;
    q += k * s;
  }
  return {X, DX};
}
pair<vector<ll>, vector<ll>> max_of_linear_segments(ll a, ll b, ll m) {
  return min_of_linear_segments(-a, m - 1 - b, m);
}
// min {x >= 0 : l <= f(x) <= r}; -1, если множество пусто.
// 0 <= l <= r < m; O(log m).
ll min_rem(ll a, ll b, ll m, ll l, ll r) {
  if (l > r) return -1;
  ll len = r - l + 1;
  b = ((b - l) % m + m) % m;
  if (b < len) return 0;
  auto [X, DX] = min_of_linear_segments(a, b, m);
  for (ll i = 0; i < DX.size(); ++i) {
    ll y2 = ((a * X[i + 1] + b) % m + m) % m;
    if (y2 >= len) continue;
    ll y1 = ((a * X[i] + b) % m + m) % m;
    ll d = (y1 - y2) / ((X[i + 1] - X[i]) / DX[i]);
    return X[i] + ((y1 - len) / d + 1) * DX[i];
  }
  return -1;
}
// {x,y}: y = min_{0 <= i < n} f(i), x = min i: f(i) = y.
// n > 0; O(log m).
pair<ll, ll> min_mod(ll n, ll a, ll b, ll m) {
  a = (a % m + m) % m, b = (b % m + m) % m;
  auto [X, DX] = min_of_linear_segments(a, b, m);
  ll x = 0;
  for (ll i = 0; i < DX.size() && X[i] < n; ++i)
    x = X[i] + (min(X[i + 1], n - 1) - X[i]) / DX[i] * DX[i];
  return {x, (a * x + b) % m};
}
// {x,y}: y = max_{0 <= i < n} f(i), x = min i: f(i) = y.
// n > 0; O(log m).
pair<ll, ll> max_mod(ll n, ll a, ll b, ll m) {
  auto [x, y] = min_mod(n, -a, m - 1 - b, m);
  return {x, m - 1 - y};
}
// {x,y}: y = min_{L <= i < R} f(i), x = min i: f(i) = y.
// L < R; O(log m).
pair<ll, ll> min_of_linear(ll L, ll R, ll a, ll b, ll m) {
  auto [x, y] = min_mod(min(R - L, m), a, a * L + b, m);
  return {L + x, y};
}
// sum_{0 <= x < n} floor((a*x+b)/m); O(log m).
ll floor_sum(ll n, ll a, ll b, ll m) {
  ll ans = 0;
  while (n) {
    ll q = a / m - (a % m < 0);
    a -= q * m;
    ans += n * (n - 1) / 2 * q;
    q = b / m - (b % m < 0);
    b -= q * m;
    ans += n * q;
    ll y = a * n + b;
    if (y < m) break;
    n = y / m, b = y % m;
    swap(a, m);
  }
  return ans;
}
// sum_{L <= x < R} floor((a*x+b)/m); O(log m).
ll floor_sum_of_linear(ll L, ll R, ll a, ll b, ll m) {
  return floor_sum(R - L, a, a * L + b, m);
}
// |{x : 0 <= x < n, l <= f(x) <= r}|.
// 0 <= l <= r < m; l > r -> 0; O(log m).
ll count_mod(ll n, ll a, ll b, ll m, ll l, ll r) {
  if (l > r) return 0;
  a = (a % m + m) % m, b = (b % m + m) % m;
  return floor_sum(n, a, b - l, m) -
         floor_sum(n, a, b - r - 1, m);
}
// sum_{0 <= x < n} f(x); O(log m).
ll sum_mod(ll n, ll a, ll b, ll m) {
  a = (a % m + m) % m, b = (b % m + m) % m;
  return n * (n - 1) / 2 * a + n * b -
         m * floor_sum(n, a, b, m);

\subsection{Interpolation}
\begin{lstlisting}[language=C++]
// for double: ll -> dbl, no %MOD, inv(x) = 1/x;
// for precision pick $x_k = c \cdot \cos \frac{k \pi}{n-1}$.
// x distinct, $P(x_i) = y_i$, $P(X) = \sum \text{res}_i X^i$
// $O(n^2)$. Mod: x, y, $\text{res} \in (-\text{MOD}, \text{MOD})$
vector<ll> interpolate(vector<ll> x, vector<ll> y) {
  int n = x.size();
  vector<ll> res(n), temp(n), d(n);
  for (int k = 0; k + 1 < n; ++k) {
    ll p = 1;
    for (int i = k + 1; i < n; ++i)
      d[i] = p, p = p * (x[i] - x[k]) % MOD;
    p = inv(p);
    for (int i = n - 1; i > k; --i) {
      y[i] = (y[i] - y[k]) * (d[i] * p % MOD) % MOD;
      p = p * (x[i] - x[k]) % MOD;
    }
  }
  ll last = 0; temp[0] = 1;
  for (int k = 0; k < n; ++k)
    for (int i = 0; i < min(n, k + 2); ++i) {
      res[i] = (res[i] + y[k] * temp[i]) % MOD;
      swap(last, temp[i]);
      temp[i] = (temp[i] - last * x[k]) % MOD;
    }
  return res;
}

ll eval(const vector<ll> &a, ll x) {
  ll r = 0;
  for (int i = ssize(a)-1; i >= 0; --i) r = (r*x+a[i]) % MOD;
  return r;
}

// $\deg P < n$ задан значениями $y_i = P(i)$, $0 \le i < n$
// Возвращает $P(X)$ за $O(n)$. $0 < n \le$ MOD, $X \in (-\text{MOD}, \text{MOD})$
// dbl: нужно $(n-1)!$ в типе: $n \le 171$ (long double: 1755)
// Если $y_i = P(x_0 + ih)$: передать $X \to (X - x_0) / h$
ll lagrange(const vector<ll> &y, ll X) {
  int n = y.size();
  vector<ll> suf(n + 1, 1);
  ll pre = 1, res = 0;
  for (int i = n; i--; ) suf[i] = suf[i + 1] * (i - X) % MOD;
  for (int i = 0; i < n; ++i) {
    auto t = pre * suf[i + 1] % MOD * ifact[i] % MOD;
    res = (res + t * ifact[n - 1 - i] % MOD * y[i]) % MOD;
    pre = pre * (X - i) % MOD;
  }
  return res;
}

\subsection{Min25}
\begin{lstlisting}[language=C++]
// Sum multiplicative f over [1..n], O(n^{3/4}/log n).
// Need f(p) = polynomial in p. Change D, POLY, sum_power(), f_prime_power().
// Default: f(n) = n, f(p) = p, f(p^e) = p^e.
// Prime count: D=0, POLY={1}, build(n), answer = prime_poly(n).
// phi: POLY={-1,1}, f_prime_power=(pe-prev+mod)%mod.
// f_prime_power: any O(1) in [0, mod), e=1 must match POLY.

// ll mul(ll a, ll b) in [0, mod); a, b may be negative!

struct Min25 {
  // f(p) = sum POLY[d] * p^d. Add degrees if needed.
  const int D = 1;
  const vector<ll> POLY = {0, 1}; // id: p

  ll n, sq;
  vector<ll> val;
  vector<int> id_small, id_large;
  vector<vector<ll>> g; // g[d][i] = sum_{p<=val[i]} p^d
  vector<ll> primes;
  bool built = false;

  int id(ll x) { return x<=sq ? id_small[x] : id_large[n/x]; }

  ll sum_power(int d, ll x) {
    x %= mod;
    if (d == 0) return x;
    if (d == 1) return mul(x, x + 1) * ((mod + 1) / 2) % mod;
    // Add interpolation here for larger D.
    assert(false);
  }

  ll prime_poly_at_index(int i) {
    ll res = 0;
    for (int d = 0; d <= D; ++d)
      res = (res + mul(POLY[d], g[d][i])) % mod;
    return res;
  }

  ll prime_poly(ll x) {
    return x < 2 ? 0 : prime_poly_at_index(id(x));
  }

  // Value f(p^e). Args are (p, e, p^e mod mod, p^{e-1} mod mod).
  ll f_prime_power(ll p, int e, ll pe, ll prev) { return pe; }

  void build(ll _n) {
    if (built) { assert(n == _n); return; }
    built = true, n = _n;
    val.clear(), primes.clear();
    sq = sqrtl(n);
    while ((sq + 1) * (sq + 1) <= n) ++sq;
    while (sq * sq > n) --sq;
    id_small.assign(sq + 1, -1);
    id_large.assign(sq + 1, -1);
    for (ll l = 1, r; l <= n; l = r + 1) {
      ll x = n / l;
      r = n / x;
      int i = val.size();
      val.pb(x);
      if (x <= sq) id_small[x] = i;
      else id_large[n / x] = i;
    }
    g.assign(D + 1, vector<ll>(val.size()));
    for (int d = 0; d <= D; ++d)
      for (int i = 0; i < val.size(); ++i)
        g[d][i] = (sum_power(d, val[i]) - 1 + mod) % mod;
    vector<ll> ppow(D + 1, 1); // ppow[d] = p^d
    for (ll p = 2; p <= sq; ++p) {
      if (g[0][id(p)] == g[0][id(p - 1)]) continue;
      primes.pb(p);
      for (int d = 1; d<=D; ++d) ppow[d] = mul(ppow[d-1], p);
      for (int i = 0; i < val.size() && val[i] >= p*p; ++i) {
        int ip = id(val[i] / p);
        int before_p = id(p - 1);
        for (int d = 0; d <= D; ++d) {
          ll sub = mul(g[d][ip] - g[d][before_p], ppow[d]);
          g[d][i] = (g[d][i] - sub + mod) % mod;
        }
      }
    }
  }

  // Sum of f(x), x <= limit, all prime divisors of x are >= primes[pos].
  ll dfs(ll limit, int pos) {
    ll last = pos ? primes[pos - 1] : 1;
    ll res = (prime_poly(limit) - prime_poly(last)) % mod;
    if (res < 0) res += mod;
    if (limit >= 2 && pos < primes.size()) {
      for (int i = pos; i < primes.size(); ++i) {
        ll p = primes[i];
        if ((__int128)p * p > limit) break;
        ll pe_int = p, pe_mod = p % mod, prev_mod = 1;
        for (int e = 1; (__int128)pe_int * p <= limit; ++e) {
          ll nxt_mod = mul(pe_mod, p);
          ll cur = f_prime_power(p, e, pe_mod, prev_mod);
          ll nxt = f_prime_power(p, e + 1, nxt_mod, pe_mod);
          res = (res + mul(cur, dfs(limit / pe_int, i + 1)) + nxt) % mod;
          prev_mod = pe_mod, pe_mod = nxt_mod;
          pe_int *= p;
        }
      }
    }
    return res;
  }

  // After build(n), returns sum_{i<=x} f(i).
  // Requires x <= sqrt(n) or x = floor(n / k) for some k.
  ll prefix_sum(ll x) {
    if (x <= 0) return 0;
    assert(built && x <= n);
    if (x > sq) assert(val[id_large[n / x]] == x);
    return (1 + dfs(x, 0)) % mod; // f(1) = 1
  }

  // O(n^{3/4}/log n)
  // Returns ans[id(x)] = sum_{i<=x} f(i) for every x in val.
  vector<ll> all_prefix_sums() {
    assert(built);
    vector<ll> prime_sum(val.size()), r(val.size()), ans(val.size());
    for (int i = 0; i < val.size(); ++i)
      prime_sum[i] = r[i] = prime_poly_at_index(i);
    for (int ui = ssize(primes) - 1; ui >= 0; --ui) {
      ll p = primes[ui];
      ll before = prime_sum[id(p)];
      for (int i = 0; i < val.size() && val[i] >= p*p; ++i) {
        ll pe_int = p, pe_mod = p % mod, prev_mod = 1;
        for (int e = 1; (__int128)pe_int * p <= val[i]; ++e) {
          ll nxt_mod = mul(pe_mod, p);
          ll cur = f_prime_power(p, e, pe_mod, prev_mod);
          ll nxt = f_prime_power(p, e + 1, nxt_mod, pe_mod);
          ll tail = r[id(val[i] / pe_int)] - before;
          if (tail < 0) tail += mod;
          r[i] = (r[i] + mul(cur, tail) + nxt) % mod;
          prev_mod = pe_mod, pe_mod = nxt_mod;
          pe_int *= p;
        }
      }
    }
    for (int i = 0; i < val.size(); ++i)
      ans[i] = (1 + r[i]) % mod; // f(1) = 1
    return ans;
  }

  ll solve(ll _n) {
    if (_n <= 0) return 0;
    build(_n); // After build(n), prime_poly(n) is sum f(p) over primes p <= n.
    return prefix_sum(n);
  }
};
\end{lstlisting}

\subsection{NTT}
\begin{lstlisting}[language=C++]
const int MOD = 998244353, G = 3; // $7 \cdot 17 \cdot 2^{23} + 1$
//const int MOD = 7340033, G = 5; // $7 \cdot 2^{20} + 1$
//const int MOD = 469762049, G = 30; // $7 \cdot  2^{26} + 1$
const int MAXLOG = 23;
int W[(1 << MAXLOG) + 10];
bool nttinit = false;
vector<int> pws;

// int add(), int sub(), int mul(), int binpow(), int inv()

void initNTT() {
  if (nttinit) return;
  nttinit = true;
  assert((MOD - 1) % (1 << MAXLOG) == 0);
  pws.pb(binpow(G, (MOD - 1) / (1 << MAXLOG)));
  for (int i = 0; i < MAXLOG - 1; ++i)
    pws.pb(mul(pws.back(), pws.back()));
  assert(pws.back() == MOD - 1);
  W[0] = 1;
  for (int i = 1; i < (1 << MAXLOG); ++i)
    W[i] = mul(W[i - 1], pws[0]);
}

void ntt(int n, vector<int> &a, bool rev) {
  initNTT();
  int lg = 31 - __builtin_clz(n);
  for (auto &el : a) el = (el % MOD + MOD) % MOD;
  vector<int> rv(n);
  for (int i = 1; i < n; ++i) {
    rv[i] = (rv[i >> 1] >> 1) ^ ((i & 1) << (lg - 1));
    if (rv[i] > i) swap(a[i], a[rv[i]]);
  }
  int num = MAXLOG - 1;
  for (int len = 1; len < n; len *= 2, --num) {
    for (int i = 0; i < n; i += 2 * len) {
      for (int j = 0; j < len; ++j) {
        int u = a[i + j];
        int v = mul(W[j << num], a[i + j + len]);
        a[i + j] = add(u, v);
        a[i + j + len] = sub(u, v);
      }
    }
  }
  if (rev) {
    int invn = binpow(n, MOD - 2);
    for (int i = 0; i < n; ++i) a[i] = mul(a[i], invn);
    reverse(a.begin() + 1, a.end());
  }
}

vector<int> conv(vector<int> a, vector<int> b) {
  if (a.empty() || b.empty()) return {};
  int lg = 32-__builtin_clz(max<int>(1,a.size()+b.size()-2));
  int n = 1 << lg;
  a.resize(n);
  b.resize(n);
  ntt(n, a, false);
  ntt(n, b, false);
  for (int i = 0; i < n; ++i) a[i] = mul(a[i], b[i]);
  ntt(n, a, true);
  while (a.size() > 1 && a.back() == 0) a.pop_back();
  return a;
}

vector<int> add(vector<int> a, vector<int> b) {
  a.resize(max(a.size(), b.size()));
  for (int i = 0; i < b.size(); ++i) a[i] = add(a[i], b[i]);
  return a;
}

vector<int> sub(vector<int> a, vector<int> b) {
  a.resize(max(a.size(), b.size()));
  for (int i = 0; i < b.size(); ++i) a[i] = sub(a[i], b[i]);
  return a;
}

vector<int> inv(const vector<int> &a, int need) {
  vector<int> b = {inv(a[0])};
  while (b.size() < need) {
    vector<int> a1 = a;
    int m = b.size();
    a1.resize(min((int) a1.size(), 2 * m));
    b = conv(b, sub({2}, conv(a1, b))), b.resize(2 * m);
  }
  b.resize(need);
  return b;
}

vector<int> div(vector<int> a, vector<int> b) {
  if (count(all(a), 0) == a.size()) return {0};
  assert(a.back() != 0 && b.back() != 0);
  int n = a.size() - 1;
  int m = b.size() - 1;
  if (n < m) return {0};
  reverse(all(a));
  reverse(all(b));
  a.resize(n - m + 1);
  b.resize(n - m + 1);
  vector<int> c = inv(b, b.size()), q = conv(a, c);
  q.resize(n - m + 1), reverse(all(q));
  return q;
}

vector<int> mod(vector<int> a, vector<int> b) {
  auto res = sub(a, conv(b, div(a, b)));
  while (res.size() > 1 && res.back() == 0) res.pop_back();
  return res;
}

vector<int> multipoint(vector<int> a, vector<int> x) {
  int n = x.size();
  vector<vector<int>> tree(2 * n);
  for (int i = 0; i < n; ++i) tree[i + n] = {x[i], MOD - 1};
  for (int i = n - 1; i; --i)
    tree[i] = conv(tree[2 * i], tree[2 * i + 1]);
  tree[1] = mod(a, tree[1]);
  for (int i = 2; i < 2 * n; ++i)
    tree[i] = mod(tree[i >> 1], tree[i]);
  vector<int> res(n);
  for (int i = 0; i < n; ++i) res[i] = tree[i + n][0];
  return res;
}

vector<int> deriv(vector<int> a) {
  for (int i = 1; i < a.size(); ++i) a[i - 1] = mul(i, a[i]);
  a.back() = 0;
  if (a.size() > 1) a.pop_back();
  return a;
}

vector<int> integ(vector<int> a) {
  a.pb(0);
  for (int i = a.size() - 1; i; --i)
    a[i] = mul(a[i - 1], inv(i));
  a[0] = 0;
  return a;
}

vector<int> log(vector<int> a, int n) {
  assert(a[0] == 1);
  auto res = integ(conv(deriv(a), inv(a, n)));
  res.resize(n);
  return res;

vector<int> exp(vector<int> a, int need) {
  assert(a[0] == 0);
  vector<int> b = {1};
  while (b.size() < need) {
    vector<int> a1 = a;
    int m = b.size();
    a1.resize(min((int) a1.size(), 2 * m));
    a1[0] = add(a1[0], 1);
    b = conv(b, sub(a1, log(b, 2 * m))), b.resize(2 * m);
  }
  b.resize(need);
  return b;
}

vector<int> power_sum_poly(int k) { // $O(k \log k)$
  vector<int> a(k + 1), p(k + 2);
  for (int j = 0; j <= k; ++j) a[j] = ifact[j + 1];
  auto c = inv(a, k + 1);
  for (int j = 0; j <= k; ++j) {
    int v = mul(fact[k], mul(c[j], ifact[k + 1 - j]));
    p[k + 1 - j] = (j & 1) ? sub(0, v) : v;
  }
  return p; // $P(n) = 1^k + ... + n^k = \sum_i p_i n^i$
}

vector<int> power_sums_inv(int n, int k) { // $O(k \log k)$

  int pw = 1, t = add(n % MOD, 1);
  for (int i = 0; i <= k; ++i) {
    pw = mul(pw, t);
    a[i] = mul(sub(pw, 1), ifact[i + 1]);
    b[i] = ifact[i + 1];
  }
  auto s = conv(a, inv(b, k + 1));
  s.resize(k + 1);
  for (int i = 0; i <= k; ++i) s[i] = mul(s[i], fact[i]);
  return s; // $s_i = 1^i+...+n^i = \sum_{x=1}^n x^i$
}

vector<int> or_conv(int n, vector<int> a, vector<int> b) { // $|a| = |b| = 2^n$

    for (int j = 0; j < (1 << n); ++j) {
      if ((j >> i) & 1) {
        a[j] = (a[j] + a[j ^ (1 << i)]) % MOD;
        b[j] = (b[j] + b[j ^ (1 << i)]) % MOD;
      }
    }
  }
  vector<int> c(1 << n);
  for (int i = 0; i < (1 << n); ++i)
    c[i] = (ll)a[i] * b[i] % MOD;
  for (int i = n - 1; i >= 0; --i) {
    for (int j = (1 << n) - 1; j >= 0; --j)
      if ((j >> i) & 1)
        c[j] = (c[j] - c[j ^ (1 << i)] + MOD) % MOD;
  }
  return c;
}

        a[j + k] = (a[j + k] + a[i + j + k]) % MOD;
        b[j + k] = (b[j + k] + b[i + j + k]) % MOD;
      }
    }

  vector<int> c(1 << n);
  for (int i = 0; i < (1 << n); ++i)

vector<int> xor_conv(int n, vector<int> a, vector<int> b) { // $|a| = |b| = 2^n$

    for (int j = 0; j < (1 << n); ++j) {
      if ((j & i) == 0) {
        int x = a[j], y = a[j | i];
        a[j] = (x + y) % MOD, a[j | i] = (x - y + MOD) % MOD;
        x = b[j], y = b[j | i];
        b[j] = (x + y) % MOD, b[j | i] = (x - y + MOD) % MOD;
      }
    }
  }
  vector<int> c(1 << n);
  for (int i = 0; i < (1 << n); ++i)
    c[i] = (ll)a[i] * b[i] % MOD;
  for (int i = 1; i < (1 << n); i *= 2) {
    for (int j = 0; j < (1 << n); ++j) {
      if ((j & i) == 0) {
        int x = c[j], y = c[j | i];
        c[j] = inv2*(x+y)%MOD, c[j | i] = inv2*(x-y+MOD)%MOD;
      }
    }
  }
  return c;

\subsection{convMod}
\begin{lstlisting}[language=C++]
// a[i] in [0, MOD), MOD <~ 1e9, n + m <= 2^20 (precision)
vector<ll> conv_mod(const vector<ll> &a, const vector<ll> &b) {
  if (a.empty() || b.empty()) return {};
  vector<ll> res(a.size() + b.size() - 1);
  int lg = 32 - __builtin_clz(max<int>(1, res.size() - 1));
  int n = 1 << lg, cut = (int) (sqrt(MOD));
  vector<comp> L(n), R(n), outs(n), outl(n);
  // x = h * cut + l, |h|, |l| <~ cut / 2: 4x less fft error
  auto sp = [&](ll x) {
    if (x > MOD / 2) x -= MOD;
    ll h = rint((double)x / cut);
    return comp(h, x - h * cut);
  };
  for (int i = 0; i < a.size(); ++i) L[i] = sp(a[i]);
  for (int i = 0; i < b.size(); ++i) R[i] = sp(b[i]);
  fft(L, lg), fft(R, lg);
  for (int i = 0; i < n; ++i) {
    int j = -i & (n - 1);
    outl[j] = (L[i] + L[j].conj()) * R[i] / (2.0 * n);
    outs[j] = (L[i] - L[j].conj()) * R[i] / (2.0 * n) * comp(0, 1) * -1;
  }
  fft(outl, lg), fft(outs, lg);
  for (int i = 0; i < res.size(); ++i) {
    ll av = rint(outl[i].x), cv = rint(outs[i].y);
    ll bv = rint(outl[i].y) + rint(outs[i].x);
    res[i] = ((av % MOD * cut + bv) % MOD * cut + cv) % MOD;
    res[i] += res[i] < 0 ? MOD : 0;
  }
  return res;
}

\begin{lstlisting}[language=C++]
// p is prime
// -1 if no solution
// $x=\text{sqrt}(a, p) \implies x^2=a$ and $(-x)^2=a$
// $O(\log n)$ if $p \equiv 3 \mod 4$ else $O(\log^2 n)$
// should be changed if const p
ll sqrt(ll a, ll p) {
  a %= p;
  if (a < 0) a += p;
  if (a == 0) return 0;
  if (binpow(a, (p - 1) / 2, p) != 1)
    return -1; // no solution
  if (p % 4 == 3) return binpow(a, (p + 1) / 4, p);
  ll s = p - 1, n = 2;
  int r = 0, m;
  while (s % 2 == 0) ++r, s /= 2;
  while (binpow(n, (p - 1) / 2, p) != p - 1) ++n;
  ll x = binpow(a, (s + 1) / 2, p);
  ll b = binpow(a, s, p), g = binpow(n, s, p);
  for (;; r = m) {
    ll t = b;
    for (m = 0; m < r && t != 1; ++m) t = t * t % p;
    if (m == 0) return x;
    ll gs = binpow(g, 1LL << (r - m - 1), p);
    g = gs * gs % p;
    x = x * gs % p;
    b = b * g % p;
  }
}

pair<int,vector<ll>> sle(vector<vector<ll>> a, vector<ll> b) {
  int n = a.size(), m = a[0].size();
  assert(n == b.size());
  for (int i = 0; i < n; ++i) a[i].pb(b[i]);
  a = gauss(a);
  vector<ll> x(m, 0);
  for (int i = n - 1; i >= 0; --i) {
    int leftmost = m;
    for (int j = 0; j < m; ++j)
      if (sign(a[i][j])) {
        leftmost = j;
        break;
      }
    if (leftmost == m && sign(a[i].back())) return {-1, {}};
    if (leftmost == m) continue;
    ll val = a[i].back();
    for (int j = m - 1; j > leftmost; --j)
      val -= a[i][j] * x[j], val %= mod;
    x[leftmost] = (val*inv(a[i][leftmost]) % mod + mod) % mod;
  }
  return {1, x};
}

// $ax+by = \pm gcd$ if $a<0$ or $b<0$
pair<int, int> ext_gcd(int a, int b) {
  int x1 = 1, y1 = 0, x2 = 0, y2 = 1;
  while (b) {
    int k = a / b;
    x1 -= x2 * k;
    y1 -= y2 * k;
    a %= b;
    swap(x1, x2), swap(y1, y2), swap(a, b);
  }
  return {x1, y1};
}

// solve $ax+by=c$ with minimum $x \geq 0$
bool cool_ext_gcd(int a, int b, int c, int &x, int &y) {
  if (b == 0) {
    y = 0;
    if (a == 0) return x = 0, c == 0;
    return x = c / a, c % a == 0 && x >= 0;
  }
  auto [x0, y0] = ext_gcd(a, b);
  int g = (ll)x0 * a + (ll)y0 * b;
  if (c % g != 0) return false;
  x = (ll)x0 * (c / g) % (b / g);
  if (x < 0) x += abs(b / g);
  y = (c - (ll)a * x) / b;
  return true;
}

// solve $w \equiv r_i \pmod {c_i}$
// assumes $c_i > 0$ and $\operatorname{lcm}(c_i)$ fits in ll
// return minimum $w \geq 0$ and common modulo
// return {-1, -1} if there is no solution
pair<ll, ll> generalized_crt(const vector<pair<int, int>> &equations) {
  ll r = 0, mod = 1;
  for (auto [ri, ci] : equations) {
    ri %= ci;
    if (ri < 0) ri += ci;
    int reduced_mod = mod % ci;
    auto [p, q] = ext_gcd(reduced_mod, ci);
    int g = (ll)p * reduced_mod + (ll)q * ci;
    ll diff = ri - r;
    if (diff % g != 0) return {-1, -1};
    ll k = (__int128)(diff / g) * p % (ci / g);
    if (k < 0) k += ci / g;
    ll new_mod = mod / g * ci;
    r = (r + (__int128)mod * k) % new_mod;
    mod = new_mod;
  }
  return {r, mod};
}

\begin{lstlisting}[language=C++]
int count_walks_1(int b1, int b2, int p, int q) {
  // counting walks from (0, 0) to (p, q)
  // each turn x += 1 or y += 1
  // without touching y = x + b1 and y = x + b2
  // b1 < 0 < b2 must hold
  // O((p + q) / (b2 - b1))
  if (min(p, q) < 0 || q - p <= b1 || q - p >= b2) return 0;
  ll ans = C(p, p + q);
  ar(2) F = {p, q}, S = {p, q};
  int cf = mod - 1;
  while (true) {
    F[1] -= b1, swap(F[0], F[1]), F[1] += b1;
    S[1] -= b2, swap(S[0], S[1]), S[1] += b2;
    swap(F, S);
    int wf = C(F[0], F[0] + F[1]);
    int ws = C(S[0], S[0] + S[1]);
    ans += (cf * (ll) ((wf + ws) % mod)) % mod;
    if (wf == 0 && ws == 0) break;
    cf = mod - cf;
  }
  return ans % mod;
}

int count_walks_2(int b1, int b2, int p, int q) {
  // counting walks from (0, 0) to (p, q)
  // each turn x += 1 and (y -= 1 or y += 1)
  // without touching y = b1 and y = b2
  // b1 < 0 < b2 must hold
  // O(p / (b2 - b1))
  if (abs(p) % 2 != abs(q) % 2) return 0;
  return count_walks_1(b1, b2, (p - q) / 2, (p + q) / 2);
}

\subsection{Симплекс}
\begin{lstlisting}[language=C++]
// $\max c^Tx: Ax \leq b, x \geq 0, A \in \mathbb{R}^{m \times n}$ ($x$ is output only)
// Returns -inf if infeasible, +inf if unbounded ($x$ is any feasible then).
// Phase 1 (finds a feasible $x$) is skipped automatically if $b\geq 0$, since then $x = 0$ is feasible. Faster.
// $O(nm)$ per pivot; usually $\approx n+m$ pivots, $2^n$ worst case.
// eps is absolute: fine for $|A|, |c| \leq 10^6$
// dbl = double is ~7x faster than long double
dbl simplex(vector<vector<dbl>> A, vector<dbl> b,
            vector<dbl> c, vector<dbl> &x) {
  const dbl eps = 1e-8, inf = 1 / 0.0;
  int m = b.size(), n = c.size();
  vector<int> B(m), N(n+1);
  iota(all(B), n), iota(all(N), 0), N[n] = -1, x.assign(n, 0);
  A.pb(c), b.pb(0);
  vector D(m+2, vector<dbl>(n+2));
  for (int i = 0; i <= m; ++i)
    copy(all(A[i]), D[i].begin()), D[i][n+1] = b[i];
  auto pivot = [&](int r, int s) {
    dbl *a = D[r].data(), t = 1 / exchange(a[s], 1);
    for (int j = 0; j <= n+1; ++j) a[j] *= t;
    for (int i = 0; i < m+2; ++i)
      if (dbl k = D[i][s]; i != r && abs(k) > eps) {
        D[i][s] = 0;
        for (int j = 0; j <= n+1; ++j) D[i][j] -= a[j] * k;
      }
    swap(B[r], N[s]);
  };
  auto run = [&](int x) {
    int deg = 0, r, s;
    auto p = [&](int j) {
      return pair(deg > n+m ? D[x][j] > eps : D[x][j], -N[j]);
    };
    auto q = [&](int i) {
      return pair(D[i][n+1] / D[i][s], B[i]);
    };
    while (1) {
      r = -1, s = 0;
      for (int j = 1; j <= n; ++j)
        if (p(j) > p(s)) s = j;
      if (D[x][s] <= eps) return 1;
      for (int i = 0; i < m; ++i)
        if (D[i][s] > eps && (r < 0 || q(i) < q(r))) r = i;
      if (r < 0) return 0;
      deg = D[r][n+1] < eps ? deg + 1 : 0;
      pivot(r, s);
    }
  };
  int r = min_element(all(b)) - b.begin();
  if (D[r][n+1] < -eps) { // can remove this block if $b\geq 0$
    for (auto &d : D) d[n] = -1;
    pivot(r, n), run(m+1);
    if (D[m+1][n+1] > eps) return -inf;
    for (int i = 0, s = 0; i < m; ++i)
      if (B[i] < 0) {
        for (int j = 1; j <= n; ++j)
          if (abs(D[i][j]) > abs(D[i][s])) s = j;
        pivot(i, s);
      }
    for (auto &d : D) d[find(all(N), -1) - N.begin()] = 0;
  }
  bool ok = run(m);
  for (int i = 0; i < m; ++i)
    if (B[i] < n) x[B[i]] = D[i][n+1];
  return ok ? -D[m][n+1] : inf;
}

// same, but x[j] < 0 is allowed if neg[j]:
// x[j] = y[j] - y[n] for them, all y >= 0
dbl simplex_neg(vector<vector<dbl>> A, vector<dbl> b,
    vector<dbl> c, vector<char> neg, vector<dbl> &x) {
  int n = c.size();
  A.pb(c);
  for (auto &r : A) {
    r.pb(0);
    for (int j = 0; j < n; ++j)
      if (neg[j]) r[n] -= r[j];
  }
  c = A.back(), A.pop_back();
  dbl res = simplex(A, b, c, x);
  for (int j = 0; j < n; ++j)
    if (neg[j]) x[j] -= x[n];
  x.pop_back();
  return res;
}
\end{lstlisting}

$x(x-1)(x-2)\cdots(x-n+1)=\sum_{k=0}^{n}s_1(n,k)x^k$

\item % Стирлинговые числа первого рода: фиксированный столбец через log/exp:
$s_1(n,k)=\frac{n!}{k!}[x^{n-k}]

$s_2(n,k)=[x^{n-k}]

\item % Стирлинговые числа второго рода: вся строка через одну свёртку:
$s_2(n,k)=[x^k]\left(\sum_{i=0}^{n}\frac{i^n}{i!}x^i\right)

vector<int> z_func(const string &s) {
  int n = s.size(), l = 0, r = 0;
  vector<int> z(n, 0);
  if (n) z[0] = n;
  for (int i = 1; i < n; ++i) {
    if (i < r) z[i] = min(z[i-l], r-i);
    while (i+z[i] < n && s[z[i]] == s[i+z[i]]) z[i]++;
    if (i+z[i] > r) l = i, r = i+z[i];

vector<int> prefix_func(const string &s) {
  int n = s.size(), a = 0;
  vector<int> pref(n, 0);
  for (int i = 1; i < n; ++i) {
    while (a > 0 && s[a] != s[i]) a = pref[a-1];
    pref[i] = a += s[i] == s[a];
  }
  return pref;
}

// d[0][i] even center (i,i+1). d[1][i] odd center (i)
// radius = $\lceil \frac{\text{len}}{2} \rceil$
array<vector<int>, 2> manacher(const string &s) {
  int n = s.size();
  array<vector<int>, 2> d;
  if (!n) return d;
  for (int z = 0; z <= 1; ++z) { //0=odd,1=even here!
    d[z].resize(n - z);
    int l = 0, r = -1;
    for (int i = z; i < n; ++i) {
      int k = i > r ? 1-z : min(d[z][l+r-i], r-i+z);
      while(i-k-z>=0 && i+k<n && s[i-k-z]==s[i+k]) ++k;
      d[z][i-z] = k--;
      if (i+k > r) l = i-k-z, r = i+k;
    }
  }
  swap(d[0], d[1]);
  return d;
}

int min_rotation(const string &s) {
  int a=0, n=s.size(); string t=s+s;
  for (int b=1; b<n; ++b)
    for (int k=0; k<n; ++k)
      if (a+k==b||t[a+k]<t[b+k]) { b+=k?k-1:0; break; }
      else if (t[a+k]>t[b+k]) { a=b; break; }
  return a;
}

vector<int> kasai(const T &s, const vector<int> &sa) {
  int n = s.size(), k=0;
  vector<int> rk(n), lcp(n-!!n);
  for (int i=0; i<n; ++i) rk[sa[i]] = i;
  for (int i=0; i<n; i++, k-=k>0) {
    if (rk[i] == n-1) continue;
    int j = sa[rk[i]+1];
    while (max(i,j)+k < n && s[i+k]==s[j+k]) k++;
    lcp[rk[i]] = k;
  }
  return lcp;
}

struct ST {
  vector<int> st, mx, mx_cnt, sec_mx;

  ST(vector<int> &a) {
    int n = a.size();
    assert(n>0);
    st.resize(n * 4), mx.resize(n * 4);
    mx_cnt.resize(n * 4, 0), sec_mx.resize(n * 4, 0);
    build(0, 0, n, a);
  }

  void upd_from_children(int v) {
    st[v] = st[v * 2 + 1] + st[v * 2 + 2];
    mx[v] = max(mx[v * 2 + 1], mx[v * 2 + 2]);
    mx_cnt[v] = 0;
    sec_mx[v] = max(sec_mx[v * 2 + 1], sec_mx[v * 2 + 2]);
    if (mx[v * 2 + 1] == mx[v])
      mx_cnt[v] += mx_cnt[v * 2 + 1];
    else
      sec_mx[v] = max(sec_mx[v], mx[v * 2 + 1]);
    if (mx[v * 2 + 2] == mx[v])
      mx_cnt[v] += mx_cnt[v * 2 + 2];
    else
      sec_mx[v] = max(sec_mx[v], mx[v * 2 + 2]);
  }

  void build(int i, int l, int r, vector<int> &a) {
    if (l + 1 == r) {
      st[i] = mx[i] = a[l];
      mx_cnt[i] = 1, sec_mx[i] = -INF;
      return;
    }
    int m = (r + l) / 2;
    build(i * 2 + 1, l, m, a);
    build(i * 2 + 2, m, r, a);
    upd_from_children(i);
  }

  void push_min_eq(int v, int val) {
    if (mx[v] > val) {
      st[v] -= (mx[v] - val) * mx_cnt[v];
      mx[v] = val;
    }
  }

  void push(int i, int l, int r) {
    if (l + 1 < r) {
      push_min_eq(i * 2 + 1, mx[i]);
      push_min_eq(i * 2 + 2, mx[i]);
    }
  }

  void update(int i, int l, int r, int ql, int qr, int val) {
    if (qr <= l || r <= ql || mx[i] <= val) return;
    if (ql <= l && r <= qr && sec_mx[i] < val) {
      push_min_eq(i, val);
      return;
    }
    push(i, l, r);
    int m = (l + r) / 2;
    update(i * 2 + 1, l, m, ql, qr, val);
    update(i * 2 + 2, m, r, ql, qr, val);
    upd_from_children(i);
  }

  int sum(int i, int l, int r, int ql, int qr) {
    if (qr <= l || r <= ql) return 0;
    push(i, l, r);
    if (ql <= l && r <= qr) return st[i];
    int m = (l + r) / 2;
    return sum(i*2+1,l,m,ql,qr) + sum(i*2+2,m,r,ql,qr);
  }

    nds[t].l = r, pull(t);
    return {l, t};
  } else {
    auto [l, r] = split(nds[t].r, k - szl - 1);
    nds[t].r = l, pull(t);
    return {t, r};
  }
}

// не забыть обновлять sz в pull

// insert $(key,val)$, erase $key$, $\max(val)$ for $key\in[l,r)$, $val$+= for $key\in[l,r)$
struct Node {
  int l, r;
  int x, y;
  int val, mx, mod;

  // value of empty set
  Node() : val(-INF), mx(-INF) { l = r = 0, mod = 0; }
  Node(int x, int val) : x(x), val(val), mx(val) {
    l = r = 0, mod = 0, y = rnd();
  }
};

Node nds[MAX];
int ndsz = 1; // nds[0] means empty

void push(int t) {
  if (!t || nds[t].mod == 0) return;
  nds[t].val += nds[t].mod;
  nds[t].mx += nds[t].mod;
  if (nds[t].l) nds[nds[t].l].mod += nds[t].mod;
  if (nds[t].r) nds[nds[t].r].mod += nds[t].mod;
  nds[t].mod = 0;
}

// delete push(t) if sure (faster)
int getmx(int t) { push(t); return nds[t].mx; }

void pull(int t) {
  if (!t) return;
  push(t), push(nds[t].l), push(nds[t].r); // must have
  nds[t].mx = max(nds[t].val, max(getmx(nds[t].l), getmx(nds[t].r)));
}

pair<int, int> split(int t, int x) {
  if (!t) return {0, 0};
  push(t);
  if (x <= nds[t].x) {
    auto [l, r] = split(nds[t].l, x);
    nds[t].l = r, pull(t);
    return {l, t};
  } else {
    auto [l, r] = split(nds[t].r, x);
    nds[t].r = l, pull(t);
    return {t, r};
  }
}

int merge(int l, int r) {
  push(l), push(r);
  if (!l) return r;
  if (!r) return l;
  if (nds[l].y < nds[r].y) {
    nds[l].r = merge(nds[l].r, r), pull(l);
    return l;
  } else {
    nds[r].l = merge(l, nds[r].l), pull(r);
    return r;
  }
}

void insert(int &root, int x, int val) {
  nds[ndsz++] = Node(x, val);
  auto [l, r] = split(root, x);
  root = merge(merge(l, ndsz - 1), r);
}

// erase all equal to x
void erase(int &root, int x) {
  auto [lm, r] = split(root, x + 1);
  auto [l, m] = split(lm, x);
  root = merge(l, r);
}

// query [l, r)
int query(int &root, int ql, int qr) {
  auto [lm, r] = split(root, qr);
  auto [l, m] = split(lm, ql);
  int res = getmx(m);
  root = merge(merge(l, m), r);
  return res;
}

// update [l, r)
void update(int &root, int ql, int qr, int qx) {
  auto [lm, r] = split(root, qr);
  auto [l, m] = split(lm, ql);
  if (m) nds[m].mod += qx;
  root = merge(merge(l, m), r);
}

struct Node {
  int l, r, val, sz, sm;
  Node() : l(0), r(0), val(0), sz(0), sm(0) {}
  Node(int val, int l, int r) : val(val), l(l), r(r) {
    sz = 1 + size(l) + size(r);
    sm = val + sum(l) + sum(r);
  }
};

Node nds[MAX];
int ndsz = 1;

int size(int t) { return nds[t].sz; }
int sum(int t) { return nds[t].sm; }

int newNode(int val, int l, int r) {
  nds[ndsz++] = Node(val, l, r);
  return ndsz - 1;
}

pair<int, int> split(int t, int x) {
  if (!t) return {0, 0};
  int szl = size(nds[t].l);
  if (szl >= x) {
    auto [l, r] = split(nds[t].l, x);
    return {l, newNode(nds[t].val, r, nds[t].r)};
  } else {
    auto [l, r] = split(nds[t].r, x - szl - 1);
    return {newNode(nds[t].val, nds[t].l, l), r};
  }
}

bool chooseleft(int szl, int szr) {
  return rnd() % (szl + szr) < szl;
}

int merge(int l, int r) {
  if (!l) return r;
  if (!r) return l;
  if (chooseleft(nds[l].sz, nds[r].sz)) {
    int nr = merge(nds[l].r, r);
    return newNode(nds[l].val, nds[l].l, nr);
  } else {
    int nl = merge(l, nds[r].l);
    return newNode(nds[r].val, nl, nds[r].r);
  }
}

int insert(int root, int pos, int val) {
  int new_v = newNode(val, 0, 0);
  auto [l, r] = split(root, pos);
  return merge(merge(l, new_v), r);
}

int erase(int root, int pos) {
  auto [lm, r] = split(root, pos + 1);
  auto [l, m] = split(lm, pos);
  return merge(l, r);
}

// query [ql, qr), no new nodes
int get(int t, int ql, int qr) {
  if (!t || qr <= 0 || size(t) <= ql) return 0;
  if (ql <= 0 && size(t) <= qr) return sum(t);
  int lsz = size(nds[t].l);
  return get(nds[t].l, ql, qr)
       + (ql <= lsz && lsz < qr ? nds[t].val : 0)
       + get(nds[t].r, ql-lsz-1, qr-lsz-1);
}

  auto a = get(nds[v].l, l, m, ql, qr);
  auto b = get(nds[v].r, m, r, ql, qr);
  return a + b;
}

struct Fenwick {
  int n;
  vector<int> f;
  Fenwick(int n) : n(n), f(n+1) {}

  void build(const vector<int> &a) { // O(n)
    n=a.size(), f.assign(n+1, 0);
    for (int i = 1; i <= n; ++i) {
      f[i] += a[i-1];
      int j = i + (i & -i);
      if (j <= n) f[j] += f[i];
    }
  }

  // a[i] += x
  void add(int i, int x) {
    for (++i; i <= n; i += i & -i) f[i] += x;
  }

  // sum a[0..i)
  int get(int i) {
    int ans = 0;
    for (; i > 0; i -= i & -i) ans += f[i];
    return ans;
  }

  // n > 0, a[..] >= 0,   pref[i] = a[0] + ... + a[i]
  // lower_bound по pref: min i, что pref[i] >= x (либо n)
  // get(lower_bound(x)) = максимальное, меньшее x (при x>0)
  int lower_bound(int x) {
    int i = 0;
    for (int b = 31 - __builtin_clz(n); b >= 0; --b) {
      int d = 1 << b;
      if (i+d <= n && f[i+d] < x) i+=d, x-=f[i];
    }
    return i;
  }
};
